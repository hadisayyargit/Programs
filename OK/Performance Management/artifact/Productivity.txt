DECLARE @Ym AS INT = 140312;

WITH ALL_cte AS (
    SELECT   
        CONCAT(LEFT([date], 4), SUBSTRING([date], 6, 2)) AS YearMonth,
        D.Value,
        D.Code, 
        CONCAT(CONCAT(LEFT([date], 4), SUBSTRING([date], 6, 2), RIGHT([date], 2)), p.CodeMask) AS Key_D_P
    FROM [HR].[Att_DailyResult] D WITH (NOLOCK)
    INNER JOIN [HR].[Prs_Person] P WITH (NOLOCK) ON P.ID = D.personelid 
    INNER JOIN [HR].[GolrangAllPerson] G WITH (NOLOCK) ON G.PersonelID = P.CodeMask
    INNER JOIN [HR].Att_Code C ON D.Code = C.CodeID
    WHERE P.companyid = 14 
      AND CONCAT(LEFT([date], 4), SUBSTRING([date], 6, 2)) = @Ym
      AND D.Code IN (14381)
      AND G.semat IN ('راننده ليفتراك','كارگر انبار','کارگرجورچين','کنترل چي')
),
Mamoriyat_cte2 AS (
    SELECT 
        D.Value,
        CONCAT(CONCAT(LEFT([date], 4), SUBSTRING([date], 6, 2), RIGHT([date], 2)), p.CodeMask) AS Key_D_P
    FROM [HR].[Att_DailyResult] D WITH (NOLOCK)
    INNER JOIN [HR].[Prs_Person] P WITH (NOLOCK) ON P.ID = D.personelid 
    INNER JOIN [HR].[GolrangAllPerson] G WITH (NOLOCK) ON G.PersonelID = P.CodeMask
    INNER JOIN [HR].Att_Code C ON D.Code = C.CodeID
    WHERE P.companyid = 14 
      AND CONCAT(LEFT([date], 4), SUBSTRING([date], 6, 2)) = @Ym
      AND D.Code IN (11032)
      AND G.semat IN ('راننده ليفتراك','كارگر انبار','کارگرجورچين','کنترل چي')
)
SELECT 
    sum(IIF(ALL_cte.Code = 14381, ALL_cte.Value - ISNULL(Mamoriyat_cte2.Value, 0), ALL_cte.Value))/60 AS Value
FROM ALL_cte
LEFT JOIN Mamoriyat_cte2 ON ALL_cte.Key_D_P = Mamoriyat_cte2.Key_D_P;
